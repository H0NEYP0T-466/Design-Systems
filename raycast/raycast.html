Developer API

Build extensions in React & TypeScript.

The Raycast API provides native UI primitives: Lists, Details, Forms, ActionPanels, and Toast notifications. Write code in modern TypeScript and run natively inside macOS.

Supported: Node.js 20 • React 18 • macOS Sandbox • Secure Storage
// raycast-extension: search-issues.tsx
import { List, ActionPanel, Action } from "@raycast/api";
import { useFetch } from "@raycast/utils";

export default function Command() {
  const { data, isLoading } = useFetch("https://api.github.com/issues");

  return (
    <List isLoading={isLoading} searchBarPlaceholder="Filter pull requests...">
      {data?.map((item) => (
        <List.Item
          key={item.id}
          title={item.title}
          accessories={[{ text: item.author }]}
          actions={
            <ActionPanel>
              <Action.OpenInBrowser url={item.html_url} />
            </ActionPanel>
          }
        />
      ))}
    </List>
  );
}
