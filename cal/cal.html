DEVELOPER INTEGRATION

Embed with the official React SDK

import Cal, { getCalApi } from "@calcom/embed-react";
import { useEffect } from "react";

export default function BookingPage() {
  useEffect(() => {
    (async function () {
      const cal = await getCalApi();
      cal("ui", { theme: "light", styles: { branding: { brandColor: "#242424" } } });
    })();
  }, []);

  return <Cal calLink="peer/30min" style={{ width: "100%", height: "100%", overflow: "scroll" }} />;
}
